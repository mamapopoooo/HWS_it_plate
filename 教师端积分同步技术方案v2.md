## 1. 核心设计原则

### 1.1 学号作为学生唯一标识

```
教师端：使用学号（如 "2026001"）
网页端：使用学号（如 "2026001"）
数据库：users.username = 学号
```

**优势：**
- 教师端已有学号数据，无需转换
- 学生容易记住自己的学号
- 与学校系统一致

### 1.2 数据驱动设计（易于扩展）

所有可配置内容都从数据库读取，不硬编码：

| 内容 | 数据库表 | 扩展方式 |
|------|----------|----------|
| 宠物种类 | `pets_template` | 直接插入新记录 |
| 技能列表 | `skills_template` | 直接插入新记录 |
| 道具列表 | `items_template` | 直接插入新记录 |
| 兑换规则 | `exchange_rules` | 直接插入新记录 |
| 积分规则 | `points_rules` | 直接插入新记录 |

**扩展流程：**
```
管理员在后台添加新宠物 → 插入数据库 → 前端自动显示（无需改代码）
```

### 1.3 登录权限分离

| 角色 | 登录方式 | 权限 |
|------|----------|------|
| 学生 | 学号 + 姓名（免密） | 查看积分、兑换道具、战斗 |
| 管理员 | 账号 + 密码 | 管理宠物/技能/道具、查看数据、配置规则 |

## 2. 数据库设计

### 2.1 用户表（users）

```sql
CREATE TABLE IF NOT EXISTS users (
  id TEXT PRIMARY KEY,              -- 内部UUID（不对外暴露）
  username TEXT UNIQUE NOT NULL,    -- 学号（学生）或工号（管理员）
  password_hash TEXT,               -- 密码哈希（学生可为空）
  role TEXT NOT NULL DEFAULT 'student',  -- student/admin
  real_name TEXT NOT NULL,          -- 真实姓名
  class_id TEXT,                    -- 班级ID（学生专用）
  grade INTEGER,                    -- 年级（6/7/8...）
  total_points INTEGER DEFAULT 0,   -- 积分总额
  created_at INTEGER,
  last_login_at INTEGER
);

-- 索引
CREATE INDEX IF NOT EXISTS idx_users_username ON users(username);
CREATE INDEX IF NOT EXISTS idx_users_class ON users(class_id);
CREATE INDEX IF NOT EXISTS idx_users_role ON users(role);
```

### 2.6 积分记录表（points_records）- 新增

```sql
CREATE TABLE IF NOT EXISTS points_records (
  id TEXT PRIMARY KEY,           -- 记录ID（教师端生成）
  student_id TEXT NOT NULL,      -- 学号（不是UUID）
  teacher_id TEXT,               -- 教师工号
  amount INTEGER NOT NULL,       -- 分值

CREATE INDEX IF NOT EXISTS idx_points_category ON points_records(category);
```

```vue
<script setup lang="ts">

import { useAuthStore } from '@/stores/auth'

const authStore = useAuthStore()
```

```vue
  // 从API动态加载道具列表

function getItemEmoji(type: string) {
  const emojis: Record<string, string> = {
    exp: '✨',
    egg: '🥚',
    reset: '🔄',
    skin: '🎨',
    boost: '⚡'
  }
  return emojis[type] || '📦'
}

async function redeemItem(item: any) {
  if (user.totalPoints < item.costPoints) {
    alert('积分不足')
    return
  }
  
  const res = await fetch('/api/game/items/redeem', {
    method: 'POST',
    headers: {
      Authorization: `Bearer ${authStore.token}`,
      'Content-Type': 'application/json'
    },
    body: JSON.stringify({ itemId: item.id })
  })
  
  const data = await res.json()
  if (data.success) {
    alert(`兑换成功！${data.result.message}`)
    user.totalPoints = data.newPoints
  } else {
    alert(`兑换失败：${data.error}`)
  }
}
```

## 5. 教师端 Python 代码更新

### 5.1 使用学号

```python
# teacher_app/points_sync.py

import json
import requests
import time
from pathlib import Path
from typing import List, Dict

class PointsSync:
    def __init__(self, api_base_url: str, teacher_token: str):
        self.api_base = api_base_url
        self.token = teacher_token
        self.data_file = Path("data/points_records.json")
        self.data_file.parent.mkdir(exist_ok=True)
        
    def add_points(self, student_number: str, amount: int, reason: str, category: str):
        """教师端加减分操作（使用学号）"""
        record = {
            "id": f"rec_{int(time.time())}_{student_number}",
            "timestamp": int(time.time() * 1000),
            "studentNumber": student_number,  # 使用学号
            "amount": amount,
            "reason": reason,
            "category": category,
            "synced": False,
            "synced_at": None
        }
        
        # 保存到本地
        self._save_to_local(record)
        
        # 尝试同步到网页端
        success = self._sync_to_server(record)
        
        if success:
            record["synced"] = True
            record["synced_at"] = int(time.time() * 1000)
            self._update_local_record(record)
        
        return success
    
    def _sync_to_server(self, record: Dict) -> bool:
        """同步单条记录到网页端API"""
        try:
            url = f"{self.api_base}/api/game/points/submit"
            headers = {
                "Authorization": f"Bearer {self.token}",
                "Content-Type": "application/json"
            }
            payload = {"records": [record]}
            
            response = requests.post(url, json=payload, headers=headers, timeout=5)
            
            if response.status_code == 200:
                result = response.json()
                return result.get("success", False)
            else:
                print(f"同步失败: {response.status_code} - {response.text}")
                return False
                
        except requests.exceptions.RequestException as e:
            print(f"网络错误: {e}")
            return False
    
    # ... 其他方法保持不变
```
